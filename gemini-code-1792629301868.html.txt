        /* --- Audio Synthesizer (Web Audio API) for Gentle Sound FX --- */
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new AudioContext();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
        }

        // Gentle, soft, child-friendly musical sounds
        function playTone(freq, type, duration, delay = 0) {
            if (isMuted) return;
            try {
                initAudio();
                setTimeout(() => {
                    const osc = audioCtx.createOscillator();
                    const gain = audioCtx.createGain();
                    
                    osc.type = type; // 'sine' or 'triangle' for warm gentle tone
                    osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
                    
                    gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                    gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
                    
                    osc.connect(gain);
                    gain.connect(audioCtx.destination);
                    
                    osc.start();
                    osc.stop(audioCtx.currentTime + duration);
                }, delay * 1000);
            } catch(e){}
        }

        function playCorrectSFX() {
            // Melodic chime (Marimba glass sound)
            playTone(523.25, 'sine', 0.2, 0);    // C5
            playTone(659.25, 'sine', 0.2, 0.1);  // E5
            playTone(783.99, 'sine', 0.25, 0.2); // G5
            playTone(1046.50, 'sine', 0.4, 0.3); // C6
        }

        function playWrongSFX() {
            // Very soft, non-scary, gentle bounce chime
            playTone(392.00, 'triangle', 0.15, 0);   // G4
            playTone(329.63, 'triangle', 0.25, 0.12); // E4
        }

        function playCleanSFX() {
            playTone(880, 'sine', 0.1, 0);

        }

        function playClickSFX() {

        function toggleMute() {
            isMuted = !isMuted;
            const icon = document.getElementById('soundIcon');
            if (isMuted) {
                icon.className = 'fa-solid fa-volume-xmark';
                if (synth) synth.cancel();
            } else {
                icon.className = 'fa-solid fa-volume-high';
                repeatSpeech();
            }
        }

        /* --- Text To Speech (TTS) Setup with Gentle Pitch --- */
        function speakText(text, onEndCallback = null) {
            if (isMuted) {
                if (onEndCallback) onEndCallback();
                return;
            }

            if (!synth) {
                if (onEndCallback) onEndCallback();
                return;
            }

            synth.cancel(); // Stop current speech

            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = 'ar-SA';
            
            // Soft & cheerful pitched voice tuning
            utterance.pitch = 1.35; // Cute, higher cartoon voice
            utterance.rate = 0.92;  // Slightly relaxed speed for clear comprehension

            // Try to set sweet female/Arabic voice if available
            const voices = synth.getVoices();
            const arVoice = voices.find(v => v.lang.includes('ar') || v.name.toLowerCase().includes('arabic') || v.name.toLowerCase().includes('laila') || v.name.toLowerCase().includes('salma') || v.name.toLowerCase().includes('zayd'));
            if (arVoice) {
                utterance.voice = arVoice;
            }

            utterance.onend = function() {
                if (onEndCallback) onEndCallback();
            };

            currentSpeechUtterance = utterance;
            synth.speak(utterance);
        }

            }
        }

            optionsContainer.innerHTML = '';

            const prefixes = ['أ', 'ب', 'ج'];
            q.options.forEach((opt, idx) => {
                const btn = document.createElement('button');

                btn.onclick = () => checkAnswer(idx, btn);
                btn.innerHTML = `
                    <span>${opt}</span>
                    <div class="opt-badge">${prefixes[idx]}</div>
                `;
                optionsContainer.appendChild(btn);
            });

            // Speak Question
            speakText(q.question);
        }

        function checkAnswer(selectedIndex, btnElement) {
            const q = questions[currentQuestionIndex];
            const allOptionBtns = document.querySelectorAll('.option-btn');

            if (selectedIndex === q.correct) {
                // Correct Answer
                playCorrectSFX();
                btnElement.classList.add('correct');
                allOptionBtns.forEach(b => b.style.pointerEvents = 'none');

                const quizBee = document.getElementById('quizBee');
                quizBee.className = "bee-wrapper bee-happy";

                honeyCount += 1;
                starCount += 2;
                updateStats();

                speakText("أحسنت! إجابة صحيحة يا بطل!", () => {
                    setTimeout(() => {
                        currentQuestionIndex++;
                        if (currentQuestionIndex === 4) {
                            // Launch Surprise Stage after Question 4
                            loadSurpriseStage();
                        } else if (currentQuestionIndex < questions.length) {
                            loadQuestion();
                        } else {
                            loadEndScreen();
                        }
                    }, 800);
                });

            } else {
                // Incorrect Answer - Gentle feedback
                playWrongSFX();
                btnElement.classList.add('wrong');
                btnElement.style.pointerEvents = 'none';

                // Show hint

                hintBox.style.display = 'flex';

                speakText("محاولة جميلة! فكّر مرة أخرى، سأساعدك بتلميح");
            }
        }
